using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using SimulatedShooting.Scene;
using TMPro;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;
using VRShooting.Application;
using VRShooting.Editor;
using VRShooting.Unity;
using VRShooting.Unity.Bootstrap;
using VRShooting.Unity.UI;

public static class MainMenuHeadingEvidence
{
    public static void CaptureBatch()
    {
        var folder = Environment.GetEnvironmentVariable("RUNTIME_GALLERY_OUT");
        Directory.CreateDirectory(folder);
        var scene = EditorSceneManager.NewPreviewScene();
        ApplicationServices services = null;
        try
        {
            var cameraRoot = new GameObject("MainMenuEvidenceCamera", typeof(Camera));
            SceneManager.MoveGameObjectToScene(cameraRoot, scene);
            var camera = cameraRoot.GetComponent<Camera>();
            camera.scene = scene;
            camera.clearFlags = CameraClearFlags.SolidColor;
            camera.backgroundColor = new Color32(7, 15, 25, 255);
            camera.orthographic = true;
            camera.nearClipPlane = .1f;
            camera.farClipPlane = 2000;
            var main = (MainMenuUI)typeof(TacticalUIGallery).GetMethod("Create", BindingFlags.NonPublic | BindingFlags.Static)
                .MakeGenericMethod(typeof(MainMenuUI)).Invoke(null, new object[] { scene, new List<GameObject>() });
            services = ApplicationServices.CreateDefault(combatScenes: new UnityCombatSceneLoader());
            main.Initialize(services);
            var root = main.gameObject;
            root.GetComponent<TrainingUICanvasAdapter>().enabled = false;
            root.GetComponent<CanvasScaler>().enabled = false;
            var canvas = root.GetComponent<Canvas>();
            canvas.renderMode = RenderMode.WorldSpace;
            canvas.worldCamera = camera;
            var rect = (RectTransform)root.transform;
            rect.sizeDelta = new Vector2(1920, 1080);
            rect.pivot = new Vector2(.5f, .5f);
            rect.position = new Vector3(960, 540, 0);
            rect.localScale = Vector3.one;
            foreach (var group in root.GetComponentsInChildren<CanvasGroup>(true)) group.alpha = 1;
            foreach (var child in root.GetComponentsInChildren<Transform>(true))
                if (child.name == "LargePanelRoot" || child.name == "MinimalHudRoot") child.gameObject.SetActive(true);
            var pages = root.GetComponentsInChildren<RectTransform>(true).Where(t => t.name.StartsWith("Screen_")).ToArray();
            foreach (var page in pages) page.gameObject.SetActive(false);
            var mainPage = pages.Single(t => t.name == "Screen_MainMenu");
            root.SetActive(true);
            mainPage.gameObject.SetActive(true);
            foreach (var text in mainPage.GetComponentsInChildren<TMP_Text>()) text.ForceMeshUpdate();
            Canvas.ForceUpdateCanvases();
            camera.orthographicSize = 540;
            camera.transform.position = new Vector3(960, 540, -1000);
            Capture(camera, Path.Combine(folder, "Screen_MainMenu.png"));
            camera.orthographicSize = 325;
            camera.transform.position = new Vector3(1555, 570, -1000);
            Capture(camera, Path.Combine(folder, "Screen_MainMenu_Closeup.png"));
            var heading = mainPage.GetComponentsInChildren<Image>().Single(i => i.name == "Panel_MainMenu_Menu");
            var label = heading.GetComponentInChildren<TMP_Text>();
            var visibleGlyphs = label.textInfo.characterInfo.Take(label.textInfo.characterCount).Count(c => c.isVisible);
            var inspection = new List<string> {
                "render=Unity Camera.Render in isolated preview scene; original UI and UI background only; no image postprocessing",
                "output=1920x1080 RGB24 PNG",
                "heading.sprite=" + heading.sprite.name,
                "heading.imageColor=" + heading.color,
                "heading.imageType=" + heading.type,
                "heading.offsetMin=" + heading.rectTransform.offsetMin,
                "heading.offsetMax=" + heading.rectTransform.offsetMax,
                "heading.outlineCount=" + heading.GetComponentsInChildren<Outline>(true).Length,
                "heading.customBorderCount=" + heading.GetComponentsInChildren<Component>(true).Count(c => c != null && c.GetType().Name == "MainMenuHeadingBorderGraphic"),
                "label.text=" + label.text,
                "label.fontSize=" + label.fontSize,
                "label.fontStyle=" + label.fontStyle,
                "label.alignment=" + label.alignment,
                "label.offsetMin=" + label.rectTransform.offsetMin,
                "label.offsetMax=" + label.rectTransform.offsetMax,
                "label.color=" + (Color32)label.color,
                "label.visibleGlyphs=" + visibleGlyphs,
                "label.overflowMode=" + label.overflowMode,
                "label.overflow=" + label.isTextOverflowing
            };
            File.WriteAllLines(Path.Combine(folder, "gallery-inspection.txt"), inspection);
            if (heading.sprite.name != "hud_capsule" || heading.color != Color.white || label.fontStyle != FontStyles.Bold || visibleGlyphs != 6)
                throw new InvalidOperationException("Main menu heading presentation contract failed");
            Debug.Log("Main menu heading evidence captured at " + folder);
        }
        finally
        {
            services?.Combat.Dispose();
            EditorSceneManager.ClosePreviewScene(scene);
        }
    }

    static void Capture(Camera camera, string path)
    {
        var target = new RenderTexture(1920, 1080, 24);
        var previous = RenderTexture.active;
        var texture = new Texture2D(1920, 1080, TextureFormat.RGB24, false);
        try
        {
            camera.targetTexture = target;
            camera.Render();
            RenderTexture.active = target;
            texture.ReadPixels(new Rect(0, 0, 1920, 1080), 0, 0);
            texture.Apply();
            File.WriteAllBytes(path, texture.EncodeToPNG());
        }
        finally
        {
            camera.targetTexture = null;
            RenderTexture.active = previous;
            UnityEngine.Object.DestroyImmediate(texture);
            target.Release();
            UnityEngine.Object.DestroyImmediate(target);
        }
    }
}
