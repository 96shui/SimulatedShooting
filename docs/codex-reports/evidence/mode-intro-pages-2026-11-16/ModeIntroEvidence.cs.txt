using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using SimulatedShooting.Scene;
using TMPro;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;
using UnityEngine.EventSystems;
using VRShooting.Application;
using VRShooting.Editor;
using VRShooting.Unity;
using VRShooting.Unity.Bootstrap;
using VRShooting.Unity.UI;

public static class ModeIntroEvidence
{
    public static void CaptureBatch()
    {
        var folder = Environment.GetEnvironmentVariable("RUNTIME_GALLERY_OUT");
        Directory.CreateDirectory(folder);
        var scene = EditorSceneManager.NewPreviewScene();
        try
        {
            var cameraRoot = new GameObject("GalleryCamera", typeof(Camera));
            SceneManager.MoveGameObjectToScene(cameraRoot, scene);
            var camera = cameraRoot.GetComponent<Camera>();
            camera.scene = scene;
            camera.clearFlags = CameraClearFlags.SolidColor;
            camera.backgroundColor = new Color32(7, 15, 25, 255);
            camera.orthographic = true;
            camera.nearClipPlane = .1f;
            camera.farClipPlane = 2000;
            var flags = BindingFlags.NonPublic | BindingFlags.Static;
            var main = (MainMenuUI)typeof(TacticalUIGallery).GetMethod("Create", flags)
                .MakeGenericMethod(typeof(MainMenuUI)).Invoke(null, new object[] { scene, new List<GameObject>() });
            main.Initialize(ApplicationServices.CreateDefault(combatScenes: new UnityCombatSceneLoader()));
            var root = main.gameObject;
            root.GetComponent<TrainingUICanvasAdapter>().enabled = false;
            root.GetComponent<CanvasScaler>().enabled = false;
            var canvas = root.GetComponent<Canvas>();
            canvas.renderMode = RenderMode.WorldSpace;
            canvas.worldCamera = camera;
            var rect = (RectTransform)root.transform;
            rect.sizeDelta = new Vector2(1920, 1080);
            rect.pivot = new Vector2(.5f, .5f);
            rect.position = new Vector3(960, 540, 0);
            rect.localScale = Vector3.one;
            foreach (var group in root.GetComponentsInChildren<CanvasGroup>(true)) group.alpha = 1;
            foreach (var child in root.GetComponentsInChildren<Transform>(true))
                if (child.name == "LargePanelRoot" || child.name == "MinimalHudRoot") child.gameObject.SetActive(true);
            var pages = root.GetComponentsInChildren<RectTransform>(true).Where(t => t.name.StartsWith("Screen_")).ToArray();
            foreach (var page in pages) page.gameObject.SetActive(false);
            var mainPage = pages.Single(t => t.name == "Screen_MainMenu");
            root.SetActive(true);
            mainPage.gameObject.SetActive(true);
            foreach (var text in mainPage.GetComponentsInChildren<TMP_Text>()) text.ForceMeshUpdate();
            Canvas.ForceUpdateCanvases();
            camera.orthographicSize = 540;
            camera.transform.position = new Vector3(960, 540, -1000);
            Capture(camera, 1920, 1080, Path.Combine(folder, "Screen_MainMenu.png"));
            camera.orthographicSize = 325;
            camera.transform.position = new Vector3(1555, 570, -1000);
            Capture(camera, 1920, 1080, Path.Combine(folder, "Screen_MainMenu_Closeup.png"));
            camera.orthographicSize = 540;
            camera.transform.position = new Vector3(960, 540, -1000);
            var buttons = mainPage.GetComponentsInChildren<Button>(true);
            var hover = buttons.Single(t => t.name == "Button_MainMenu_OpenZeroing");
            ExecuteEvents.Execute(hover.gameObject, new PointerEventData(EventSystem.current), ExecuteEvents.pointerEnterHandler);
            Capture(camera, 1920, 1080, Path.Combine(folder, "Screen_MainMenu_Hover.png"));
            var hoveredSprite = hover.image.overrideSprite.name;
            ExecuteEvents.Execute(hover.gameObject, new PointerEventData(EventSystem.current), ExecuteEvents.pointerExitHandler);
            var restoredSprite = hover.image.overrideSprite.name;
            File.WriteAllText(Path.Combine(folder, "hover-sprite-state.txt"), "pointerEnter=" + hoveredSprite + "\npointerExit=" + restoredSprite);
            var inspection = new List<string>();
            foreach (var entry in new[] { "OpenZeroing", "OpenMovingTarget", "Trench", "Urban" })
            {
                var mode = entry == "OpenZeroing" ? "Zeroing" : entry == "OpenMovingTarget" ? "MovingTarget" : entry;
                buttons.Single(t => t.name == "Button_MainMenu_" + entry).onClick.Invoke();
                var intro = pages.Single(t => t.name == "Screen_ModeIntro_" + mode);
                var scroll = intro.GetComponentInChildren<ScrollRect>();
                LayoutRebuilder.ForceRebuildLayoutImmediate(scroll.content);
                Canvas.ForceUpdateCanvases();
                foreach (var label in intro.GetComponentsInChildren<TMP_Text>())
                {
                    label.ForceMeshUpdate();
                    inspection.Add(label.name + " text=" + label.text.Replace("\n", " | ") + " font=" + label.fontSize +
                        " rect=" + label.rectTransform.rect + " preferredHeight=" + label.preferredHeight + " overflow=" + label.isTextOverflowing);
                }
                Capture(camera, 1920, 1080, Path.Combine(folder, "Screen_ModeIntro_" + mode + ".png"));
                inspection.Add(mode + " viewport=" + scroll.viewport.rect + " content=" + scroll.content.rect + " horizontal=" + scroll.horizontal);
                buttons.Single(t => t.name == "Button_ModeIntro_" + mode + "_Back").onClick.Invoke();
            }
            foreach (var part in mainPage.GetComponentsInChildren<RectTransform>(true).Where(t => t.name == "Panel_MainMenu_Menu" || t.name.Contains("_Number") || t.name.Contains("IconBounds")))
                inspection.Add(part.name + " min=" + part.offsetMin + " max=" + part.offsetMax + " sprite=" + part.GetComponent<Image>()?.sprite?.name + " color=" + part.GetComponent<Image>()?.color);
            File.WriteAllLines(Path.Combine(folder, "gallery-inspection.txt"), inspection);
            Debug.Log("Mode introduction UI evidence captured.");
        }
        finally { EditorSceneManager.ClosePreviewScene(scene); }
    }

    static void Capture(Camera camera, int width, int height, string path)
    {
        var target = new RenderTexture(width, height, 24);
        var previous = RenderTexture.active;
        var texture = new Texture2D(width, height, TextureFormat.RGB24, false);
        try
        {
            camera.targetTexture = target;
            camera.Render();
            RenderTexture.active = target;
            texture.ReadPixels(new Rect(0, 0, width, height), 0, 0);
            texture.Apply();
            File.WriteAllBytes(path, texture.EncodeToPNG());
        }
        finally
        {
            camera.targetTexture = null;
            RenderTexture.active = previous;
            UnityEngine.Object.DestroyImmediate(texture);
            target.Release();
            UnityEngine.Object.DestroyImmediate(target);
        }
    }
}
